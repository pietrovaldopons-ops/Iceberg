function headersSupabase(usaTokenUtente = true) {
  const bearer = (usaTokenUtente && accessTokenCorrente) ? accessTokenCorrente : SUPABASE_CONFIG.publishableKey;
  return {
    'apikey': SUPABASE_CONFIG.publishableKey,
    'Authorization': 'Bearer ' + bearer,
    'Content-Type': 'application/json',
    'Prefer': 'return=representation'
  };
}

function normalizzaArticolo(a) {
  const totale = Number(a.letture) || 1;
  const oggi = a.letture_oggi !== undefined ? Number(a.letture_oggi) : Math.max(1, Math.round(totale * 0.2));
  const setteGg = a.letture_7g !== undefined ? Number(a.letture_7g) : Math.max(oggi, Math.round(totale * 0.6));
  const trentaGg = a.letture_30g !== undefined ? Number(a.letture_30g) : totale;
  const username = a.autore_username || String(a.autore || 'autore').toLowerCase().replace(/[^a-z0-9_]+/g, '_');
  return {
    ...a,
    categoria: String(a.categoria || 'CRONACA').trim().toUpperCase(),
    autore_username: username,
    letture: totale,
    letture_oggi: oggi,
    letture_7g: setteGg,
    letture_30g: trentaGg
  };
}

function salvaCacheLocale(articoli, categorie, stanze, utenti) {
  try {
    if (articoli) localStorage.setItem(SUPABASE_CONFIG.cacheArticoli, JSON.stringify(articoli));
    if (categorie) localStorage.setItem(SUPABASE_CONFIG.cacheCategorie, JSON.stringify(categorie));
    if (stanze) localStorage.setItem(SUPABASE_CONFIG.cacheStanze, JSON.stringify(stanze));
    if (utenti) localStorage.setItem(SUPABASE_CONFIG.cacheUtenti, JSON.stringify(utenti));
  } catch (e) {}
}

function leggiCacheLocale() {
  try {
    const artRaw = JSON.parse(localStorage.getItem(SUPABASE_CONFIG.cacheArticoli) || 'null');
    const cat = JSON.parse(localStorage.getItem(SUPABASE_CONFIG.cacheCategorie) || '[]');
    const stanze = JSON.parse(localStorage.getItem(SUPABASE_CONFIG.cacheStanze) || '[]');
    const utentiRaw = JSON.parse(localStorage.getItem(SUPABASE_CONFIG.cacheUtenti) || 'null');
    const sessioneAuth = JSON.parse(localStorage.getItem(SUPABASE_CONFIG.sessioneAuth) || 'null');
    const sbloccate = JSON.parse(localStorage.getItem(SUPABASE_CONFIG.cacheStanzeSbloccate) || '[]');
    return {
      art: Array.isArray(artRaw) && artRaw.length ? artRaw.map(normalizzaArticolo) : null,
      cat,
      stanze,
      utenti: Array.isArray(utentiRaw) && utentiRaw.length ? utentiRaw : UTENTI_INIZIALI,
      sessioneAuth,
      sbloccate
    };
  } catch (e) {
    return { art: null, cat: [], stanze: [], utenti: UTENTI_INIZIALI, sessioneAuth: null, sbloccate: [] };
  }
}

function salvaSessioneAuth(sessione) {
  try {
    if (sessione) {
      localStorage.setItem(SUPABASE_CONFIG.sessioneAuth, JSON.stringify(sessione));
    } else {
      localStorage.removeItem(SUPABASE_CONFIG.sessioneAuth);
    }
  } catch (e) {}
}

function salvaStanzeSbloccate(listaNomi) {
  try {
    localStorage.setItem(SUPABASE_CONFIG.cacheStanzeSbloccate, JSON.stringify(Array.from(new Set(listaNomi))));
  } catch (e) {}
}

function pulisciUsername(rawUsername, fallbackEmail = '') {
  const base = rawUsername || (fallbackEmail ? fallbackEmail.split('@')[0] : 'autore');
  const clean = String(base).trim().toLowerCase().replace(/[^a-z0-9_]+/g, '_').replace(/^_+|_+$/g, '');
  return clean.length >= 2 ? clean : 'autore_iceberg';
}

    try {
      const [resArt, resCat, resStanze, resUtenti] = await Promise.all([
        fetch(`${SUPABASE_CONFIG.url}/rest/v1/articoli?select=*&order=created_at.desc,id.desc`, {
          headers: headersSupabase(false)
        }),
        fetch(`${SUPABASE_CONFIG.url}/rest/v1/categorie?select=nome&order=nome.asc`, {
          headers: headersSupabase(false)
        }),
        fetch(`${SUPABASE_CONFIG.url}/rest/v1/stanze?select=*&order=created_at.desc,id.desc`, {
          headers: headersSupabase(false)
        }),
        fetch(`${SUPABASE_CONFIG.url}/rest/v1/utenti?select=username,bio`, {
          headers: headersSupabase(false)
        })
      ]);

      if (!resArt.ok) throw new Error('Errore HTTP articoli ' + resArt.status);

      const datiArt = await resArt.json();
      const datiCat = resCat.ok ? await resCat.json() : [];
      const datiStanze = resStanze.ok ? await resStanze.json() : [];
      const datiUtenti = resUtenti.ok ? await resUtenti.json() : [];

      const articoli = Array.isArray(datiArt) && datiArt.length
        ? datiArt.map(normalizzaArticolo)
        : (cache.art || ARTICOLI_INIZIALI.map(normalizzaArticolo));

      const categorie = Array.from(new Set([
        ...CATEGORIE_BASE,

      });
      if (!res.ok) throw new Error('Errore salvataggio stanza ' + res.status);
      const creati = await res.json();
      return Array.isArray(creati) && creati[0] ? creati[0] : { ...payload, id: Date.now() };
    } catch (err) {
      return { ...payload, id: Date.now() };
    }
  },

  async creaCategoria(nome) {
    const pulita = String(nome || '').trim().toUpperCase();
    if (!pulita) return;
    try {
      await fetch(`${SUPABASE_CONFIG.url}/rest/v1/categorie`, {
        method: 'POST',
        headers: {
          ...headersSupabase(true),
          'Prefer': 'resolution=ignore-duplicates,return=representation'
        },
        body: JSON.stringify({ nome: pulita })
      });
    } catch (e) {}
  },

  async incrementaLettura(articolo) {
    if (!articolo) return;
    const norm = normalizzaArticolo(articolo);
    norm.letture += 1;
    norm.letture_oggi += 1;
    norm.letture_7g += 1;
    norm.letture_30g += 1;

    articolo.letture = norm.letture;
    articolo.letture_oggi = norm.letture_oggi;
    articolo.letture_7g = norm.letture_7g;
    articolo.letture_30g = norm.letture_30g;

    if (typeof articolo.id === 'number') {
      try {
        await fetch(`${SUPABASE_CONFIG.url}/rest/v1/articoli?id=eq.${encodeURIComponent(articolo.id)}`, {
          method: 'PATCH',
          headers: headersSupabase(false),
          body: JSON.stringify({
            letture: norm.letture,
            letture_oggi: norm.letture_oggi,
            letture_7g: norm.letture_7g,
            letture_30g: norm.letture_30g
          })
        });
      } catch (e) {}
    }
  }
};

function escapeHtml(testo) {
  return String(testo || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function getListaUtentiCompleta() {
  const mappa = new Map();
  stato.utenti.forEach(u => {
    if (u && u.username) mappa.set(u.username.toLowerCase(), u);
  });
  stato.articoli.forEach(a => {
    const uname = (a.autore_username || String(a.autore || '').toLowerCase().replace(/[^a-z0-9_]+/g, '_')).trim();
    if (uname && !mappa.has(uname)) {
      mappa.set(uname, {
        username: uname,
        bio: `Autore attivo su ICEBERG (${a.autore || uname}).`
      });
    }
  });
  return Array.from(mappa.values());
}

function richiediAccessoStanza(indiceStanza) {
  const stanza = stato.stanze[indiceStanza];
  if (!stanza) return;

  if (stato.stanzeSbloccate.includes(stanza.nome)) {
    entraInStanza(stanza);
    return;
  }

  stato.stanzaInAccesso = stanza;
  document.getElementById('titoloAccessoStanza').textContent = `Stanza: ${stanza.nome}`;
  document.getElementById('descAccessoStanza').textContent = stanza.descrizione || 'Inserisci la password per accedere al progetto comune.';
  document.getElementById('inputPasswordAccesso').value = '';
  document.getElementById('errorePasswordStanza').style.display = 'none';
  document.getElementById('modaleAccessoStanza').classList.add('aperto');
}

document.getElementById('formAccessoStanza').addEventListener('submit', e => {
  e.preventDefault();
  const stanza = stato.stanzaInAccesso;
  if (!stanza) return;
  const pwd = document.getElementById('inputPasswordAccesso').value;
  if (pwd === stanza.password_stanza) {
    if (!stato.stanzeSbloccate.includes(stanza.nome)) {
      stato.stanzeSbloccate.push(stanza.nome);
      salvaStanzeSbloccate(stato.stanzeSbloccate);
    }
    stato.stanzaInAccesso = null;
    chiudiModali();
    entraInStanza(stanza);
  } else {
    document.getElementById('errorePasswordStanza').style.display = 'block';
  }
});

function entraInStanza(stanza) {
  stato.stanzaAttiva = stanza.nome;
  stato.categoriaAttiva = 'TUTTI';

  stato.stepApprofondimento = 1;
  pulisciRicerca();

  renderArticolo(true);
}

function apriArticoloDaRicerca(idArticolo) {
  const art = stato.articoli.find(item => String(item.id) === String(idArticolo));
  if (!art) return;

  if (art.stanza_nome && !stato.stanzeSbloccate.includes(art.stanza_nome)) {
    const stanza = stato.stanze.find(s => s.nome === art.stanza_nome);
    if (stanza) {
      richiediAccessoStanza(stato.stanze.indexOf(stanza));
      return;
    }
  }

  stato.stanzaAttiva = art.stanza_nome || '';
  stato.categoriaAttiva = 'TUTTI';
  const lista = articoliFiltrati();
  const idx = lista.findIndex(item => String(item.id) === String(idArticolo));
  stato.indiceArticolo = idx >= 0 ? idx : 0;
  stato.stepApprofondimento = 1;
  pulisciRicerca();
  renderCategorie();
  renderArticolo(true);
}

  renderRisultatiRicerca();
});

  if (stato.stepApprofondimento !== 1 || !e.changedTouches.length) return;
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;
  if (Math.abs(dx) > 55 && Math.abs(dx) > Math.abs(dy) * 1.4) {
    if (dx < 0) cambiaArticolo(1);
    else cambiaArticolo(-1);
  }
}, { passive: true });

function cambiaTabAuth(tab) {
  const isLogin = tab === 'login';
  document.getElementById('btnTabLogin').classList.toggle('attivo', isLogin);
  document.getElementById('btnTabRegister').classList.toggle('attivo', !isLogin);
  document.getElementById('formLogin').style.display = isLogin ? 'block' : 'none';
  document.getElementById('formRegister').style.display = isLogin ? 'none' : 'block';
  document.getElementById('erroreAuth').style.display = 'none';
  document.getElementById('erroreRegister').style.display = 'none';
}

function apriModaleAccount(tab = 'login', azioneDopo = null) {
  stato.azioneDopoLogin = azioneDopo;
  const loggatoBox = document.getElementById('pannelloUtenteLoggato');
  const authBox = document.getElementById('pannelloAuthForms');

  if (stato.utenteCorrente) {
    loggatoBox.style.display = 'block';
    authBox.style.display = 'none';
    document.getElementById('mioAvatar').textContent = stato.utenteCorrente.username.charAt(0).toUpperCase();
    document.getElementById('mioUsername').textContent = `@${stato.utenteCorrente.username}`;
    document.getElementById('miaEmail').textContent = stato.utenteCorrente.email || '';
    document.getElementById('inputMiaBio').value = stato.utenteCorrente.bio || '';
  } else {
    loggatoBox.style.display = 'none';
    authBox.style.display = 'block';
    cambiaTabAuth(tab);
  }

  document.getElementById('modaleAccount').classList.add('aperto');
}

function completaAccessoUtente(utente) {
  stato.utenteCorrente = utente;
  if (!stato.utenti.some(u => u.username.toLowerCase() === utente.username.toLowerCase())) {
    stato.utenti.unshift({ username: utente.username, bio: utente.bio || '' });
  }
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  aggiornaBottoneAccount();
  const azione = stato.azioneDopoLogin;
  stato.azioneDopoLogin = null;
  chiudiModali();
  if (typeof azione === 'function') azione();
}

document.getElementById('formLogin').addEventListener('submit', async e => {
  e.preventDefault();
  const email = document.getElementById('loginEmail').value.trim().toLowerCase();
  const password = document.getElementById('loginPassword').value;
  const errBox = document.getElementById('erroreAuth');
  const btn = document.getElementById('btnSubmitLogin');

  btn.disabled = true;
  btn.textContent = 'Accesso...';
  errBox.style.display = 'none';

  const res = await BackendSupabase.loginConAuth({ email, password });

  btn.disabled = false;
  btn.textContent = 'Accedi';

  if (!res.ok) {
    errBox.textContent = res.errore;
    errBox.style.display = 'block';
    return;
  }

  e.target.reset();
  completaAccessoUtente(res.utente);
});

document.getElementById('formRegister').addEventListener('submit', async e => {
  e.preventDefault();
  const username = pulisciUsername(document.getElementById('regUsername').value, document.getElementById('regEmail').value);
  const email = document.getElementById('regEmail').value.trim().toLowerCase();
  const password = document.getElementById('regPassword').value;
  const bio = document.getElementById('regBio').value.trim();
  const errBox = document.getElementById('erroreRegister');
  const btn = document.getElementById('btnSubmitRegister');

  if (password.length < 6) {
    errBox.style.color = 'var(--allerta)';
    errBox.textContent = 'Inserisci una password di almeno 6 caratteri.';
    errBox.style.display = 'block';
    return;
  }

  btn.disabled = true;
  btn.textContent = 'Registrazione...';
  errBox.style.display = 'none';

  const res = await BackendSupabase.registraConAuth({ username, email, password, bio });

  btn.disabled = false;
  btn.textContent = 'Registrati con Supabase Auth';

  if (!res.ok) {
    errBox.style.color = 'var(--allerta)';
    errBox.textContent = res.errore;
    errBox.style.display = 'block';
    return;
  }

  if (!stato.utenti.some(u => u.username.toLowerCase() === res.utente.username.toLowerCase())) {
    stato.utenti.unshift({ username: res.utente.username, bio: res.utente.bio || '' });
    salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  }

  e.target.reset();

  if (res.richiedeConfermaEmail) {
    errBox.style.color = 'var(--ok)';
    errBox.textContent = 'Account creato su Supabase Auth. Controlla la tua email per confermarlo, poi accedi da "Accedi".';
    errBox.style.display = 'block';
    return;
  }

  completaAccessoUtente(res.utente);
});

async function salvaBioProfilo() {
  if (!stato.utenteCorrente) return;
  const nuovaBio = document.getElementById('inputMiaBio').value.trim();
  stato.utenteCorrente.bio = nuovaBio;
  const target = stato.utenti.find(u => u.username.toLowerCase() === stato.utenteCorrente.username.toLowerCase());
  if (target) target.bio = nuovaBio;
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  await BackendSupabase.aggiornaBio(stato.utenteCorrente.username, nuovaBio);
  chiudiModali();
}

async function effettuaLogout() {
  await BackendSupabase.logoutConAuth();
  stato.utenteCorrente = null;
  aggiornaBottoneAccount();
  chiudiModali();
}

function apriModaleAutore() {
  const contenitore = document.getElementById('listaStatistiche');
  const kpiBox = document.getElementById('kpiAutore');

  if (!stato.utenteCorrente || !stato.utenteCorrente.username) {
    kpiBox.innerHTML = '';
    contenitore.innerHTML = `
      <div class="stat-card">
        <strong>Accesso richiesto</strong>
        <p style="font-size:0.86rem; color:var(--grigio); margin:6px 0 12px;">
          Accedi al tuo account per visualizzare le statistiche degli articoli che hai scritto tu.
        </p>
        <button type="button" class="btn btn-primario" onclick="chiudiModali(); apriModaleAccount('login', apriModaleAutore);">
          Accedi ora
        </button>
      </div>
    `;
    document.getElementById('modaleAutore').classList.add('aperto');
    return;
  }

  const mioUsername = String(stato.utenteCorrente.username).trim().toLowerCase();

  const arts = stato.articoli
    .map(normalizzaArticolo)
    .filter(a => {
      const autUser = String(a.autore_username || '').trim().toLowerCase();
      const autNome = String(a.autore || '').trim().toLowerCase().replace(/^@/, '');
      return autUser === mioUsername || autNome === mioUsername;
    });

  const totOggi = arts.reduce((acc, a) => acc + a.letture_oggi, 0);
  const tot7g = arts.reduce((acc, a) => acc + a.letture_7g, 0);
  const tot30g = arts.reduce((acc, a) => acc + a.letture_30g, 0);
  const totStorico = arts.reduce((acc, a) => acc + a.letture, 0);
  const maxLetture = Math.max(1, ...arts.map(a => a.letture));

  kpiBox.innerHTML = `
    <div class="kpi-box">
      <div class="kpi-label">Oggi (@${escapeHtml(mioUsername)})</div>
      <div class="kpi-valore">${totOggi}</div>
    </div>
    <div class="kpi-box">
      <div class="kpi-label">Ultimi 7 gg</div>
      <div class="kpi-valore">${tot7g}</div>
    </div>
    <div class="kpi-box">
      <div class="kpi-label">Ultimi 30 gg</div>
      <div class="kpi-valore">${tot30g}</div>
    </div>
    <div class="kpi-box">
      <div class="kpi-label">Totale</div>
      <div class="kpi-valore">${totStorico}</div>
    </div>
  `;

  if (!arts.length) {
    contenitore.innerHTML = `
      <div class="stat-card">
        <strong>Nessun articolo ancora pubblicato da @${escapeHtml(mioUsername)}</strong>
        <p style="font-size:0.84rem; color:var(--grigio); margin:6px 0 0;">
          Quando pubblicherai un tuo articolo con "+ Crea", le sue letture compariranno qui.
        </p>
      </div>
    `;
  } else {
    contenitore.innerHTML = arts.map(a => {
      const perc = Math.min(100, Math.round((a.letture / maxLetture) * 100));
      return `
        <div class="stat-card">
          <div style="display:flex; justify-content:space-between; gap:10px; align-items:flex-start;">
            <div>
              <strong>${escapeHtml(a.titolo)}</strong><br>
              <small>${escapeHtml(a.categoria)} • @${escapeHtml(a.autore_username)}${a.stanza_nome ? ' • Stanza: ' + escapeHtml(a.stanza_nome) : ''}</small>
            </div>
            <span class="badge-cat">${a.letture} tot</span>
          </div>
          <div class="finestre-letture">
            <div class="finestra-pill"><span>Oggi</span><strong>${a.letture_oggi}</strong></div>
            <div class="finestra-pill"><span>7 giorni</span><strong>${a.letture_7g}</strong></div>
            <div class="finestra-pill"><span>30 giorni</span><strong>${a.letture_30g}</strong></div>
            <div class="finestra-pill"><span>Storico</span><strong>${a.letture}</strong></div>
          </div>
          <div class="barra-progresso"><div style="width:${perc}%"></div></div>
        </div>
      `;
    }).join('');
  }

  document.getElementById('modaleAutore').classList.add('aperto');
}

function chiudiModali() {
  document.querySelectorAll('.modale-sfondo').forEach(m => m.classList.remove('aperto'));
}

async function aggiungiCategoriaPrompt() {
  const nome = prompt('Nome nuova categoria:');
  if (!nome) return;
  const pulita = nome.trim().toUpperCase();
  if (!pulita) return;
  if (!stato.categorie.includes(pulita)) {
    stato.categorie.push(pulita);
  }
  await BackendSupabase.creaCategoria(pulita);
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  renderCategorie();
  document.getElementById('inputCategoria').value = pulita;
}

function collegaDropzone(dropEl, inputFileEl, inputUrlEl, onCaricata) {
  function leggiFile(file) {
    if (!file || !file.type.startsWith('image/')) return;
    const reader = new FileReader();
    reader.onload = ev => {
      onCaricata(ev.target.result, file.name);
      if (inputUrlEl) inputUrlEl.value = '';
    };
    reader.readAsDataURL(file);
  }

  dropEl.addEventListener('dragover', e => {
    e.preventDefault();
    dropEl.classList.add('dragover');
  });
  dropEl.addEventListener('dragleave', () => dropEl.classList.remove('dragover'));
  dropEl.addEventListener('drop', e => {
    e.preventDefault();
    dropEl.classList.remove('dragover');
    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
      leggiFile(e.dataTransfer.files[0]);
    }
  });
  inputFileEl.addEventListener('change', e => {
    if (e.target.files && e.target.files[0]) {
      leggiFile(e.target.files[0]);
    }
  });
}

function valutaControlloAi() {
  const testoCompleto = [
    document.getElementById('inputTitolo').value,
    document.getElementById('inputLiv1').value,
    document.getElementById('inputLiv2').value,
    document.getElementById('inputLiv3').value
  ].join(' ').toLowerCase();

  const box = document.getElementById('boxControlloAi');
  if (!testoCompleto.trim()) {
    box.className = 'avviso-ai';
    return;
  }

  const indiziAi = [
    'in conclusione',
    'è fondamentale notare',
    'in un mondo in rapida evoluzione',
    'spero che questo possa aiutarti',
    'come modello linguistico',
    'non esitare a chiedere'
  ];
  const trovati = indiziAi.filter(frase => testoCompleto.includes(frase));

  if (trovati.length > 0) {
    box.className = 'avviso-ai allerta';
    box.textContent = `Controllo anti-AI: rilevate formule tipiche da assistente (${trovati.join(', ')}). Riscrivi in modo più naturale e personale.`;
  } else {
    box.className = 'avviso-ai ok';
    box.textContent = 'Controllo anti-AI: nessun pattern tipico rilevato.';
  }
}

['inputTitolo', 'inputLiv1', 'inputLiv2', 'inputLiv3'].forEach(id => {
  document.getElementById(id).addEventListener('input', valutaControlloAi);
});

document.getElementById('formArticolo').addEventListener('submit', async e => {
  e.preventDefault();
  if (!stato.utenteCorrente) {
    chiudiModali();
    apriModaleAccount('login');
    return;
  }

  const urlInserito = document.getElementById('inputImmagineUrl').value.trim();
  const immagineFinale = urlInserito || stato.immagineArticoloDataUrl || IMMAGINE_DEFAULT;
  const stanzaScelta = document.getElementById('inputStanzaArticolo').value;

  const btnPubblica = document.getElementById('btnPubblica');
  btnPubblica.disabled = true;
  btnPubblica.textContent = 'Salvataggio...';

  const creato = await BackendSupabase.creaArticolo({
    categoria: document.getElementById('inputCategoria').value,
    stanza_nome: stanzaScelta,
    autore: stato.utenteCorrente.username,
    autore_username: stato.utenteCorrente.username,
    titolo: document.getElementById('inputTitolo').value,
    immagine: immagineFinale,
    livello1: document.getElementById('inputLiv1').value,
    livello2: document.getElementById('inputLiv2').value,
    livello3: document.getElementById('inputLiv3').value
  });

  stato.articoli.unshift(creato);
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  stato.stanzaAttiva = stanzaScelta || '';
  stato.categoriaAttiva = 'TUTTI';
  stato.indiceArticolo = 0;
  stato.stepApprofondimento = 1;
  stato.immagineArticoloDataUrl = '';
  document.getElementById('dropzoneImg').textContent = 'Trascina qui un\'immagine o tocca per sceglierla dal dispositivo';
  e.target.reset();

  btnPubblica.disabled = false;
  btnPubblica.textContent = 'Pubblica sul giornale';
  chiudiModali();
  renderCategorie();
  renderArticolo(false);
  if (stato.queryRicerca) renderRisultatiRicerca();
});

document.getElementById('formStanza').addEventListener('submit', async e => {
  e.preventDefault();
  if (!stato.utenteCorrente) {
    chiudiModali();
    apriModaleAccount('login');
    return;
  }

  const nomeStanza = document.getElementById('inputNomeStanza').value.trim();
  const passwordStanza = document.getElementById('inputPasswordStanza').value;
  if (!nomeStanza || !passwordStanza) return;

  const urlStanza = document.getElementById('inputImmagineStanzaUrl').value.trim();
  const immagineStanza = urlStanza || stato.immagineStanzaDataUrl || IMMAGINE_DEFAULT;

  const btnCrea = document.getElementById('btnCreaStanza');
  btnCrea.disabled = true;
  btnCrea.textContent = 'Creazione...';

  const creata = await BackendSupabase.creaStanza({
    nome: nomeStanza,
    password_stanza: passwordStanza,
    creata_da: `@${stato.utenteCorrente.username}`,
    descrizione: document.getElementById('inputDescrizioneStanza').value,
    immagine: immagineStanza
  });

  stato.stanze.unshift(creata);
  if (!stato.stanzeSbloccate.includes(creata.nome)) {
    stato.stanzeSbloccate.push(creata.nome);
    salvaStanzeSbloccate(stato.stanzeSbloccate);
  }
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);

  stato.stanzaAttiva = creata.nome;
  stato.categoriaAttiva = 'TUTTI';
  stato.indiceArticolo = 0;
  stato.stepApprofondimento = 1;
  stato.immagineStanzaDataUrl = '';
  document.getElementById('dropzoneStanzaImg').textContent = 'Trascina qui un\'immagine per la stanza o tocca per sceglierla';
  e.target.reset();

  btnCrea.disabled = false;
  btnCrea.textContent = 'Crea stanza';
  chiudiModali();
  renderCategorie();
  renderArticolo(false);
  if (stato.queryRicerca) renderRisultatiRicerca();
});
