function normalizzaPerControllo(testo) {
  return String(testo || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/0/g, 'o')
    .replace(/1/g, 'i')
    .replace(/3/g, 'e')
    .replace(/4/g, 'a')
    .replace(/@/g, 'a')
    .replace(/\$/g, 's')
    .replace(/[^a-z0-9\s]+/g, ' ');
}

function trovaParoleVietate(testo) {
  const pulito = normalizzaPerControllo(testo);
  if (!pulito.trim()) return [];
  const token = pulito.split(/\s+/).filter(Boolean);
  const trovate = new Set();

  PAROLE_VIETATE_RADICI.forEach(parola => {
    const pNorm = normalizzaPerControllo(parola);
    if (token.some(t => t === pNorm || (t.length >= pNorm.length && t.startsWith(pNorm)))) {
      trovate.add(parola);
    }
  });

  return Array.from(trovate);
}

    'Prefer': 'return=representation'
  };
}

function normalizzaStatoArticolo(rawStato) {
  const s = String(rawStato || 'pubblicato').trim().toLowerCase();
  if (s === 'bozza' || s === 'draft') return 'bozza';
  if (s === 'archiviato' || s === 'archived') return 'archiviato';
  return 'pubblicato';
}

function normalizzaArticolo(a) {
  const totale = Number(a.letture) || 0;

  const catRaw = String(a.categoria || 'CRONACA').trim().toUpperCase();

    if (stanza.nome) {
      try {
        await fetch(`${SUPABASE_CONFIG.url}/rest/v1/articoli?stanza_nome=eq.${encodeURIComponent(stanza.nome)}`, {
          method: 'PATCH',
          headers: headersSupabase(true),
          body: JSON.stringify({ stanza_nome: '' })
        });
      } catch (e) {}
    }
  },

  async creaCategoria(nome) {
    const pulita = String(nome || '').trim().toUpperCase();
    if (!pulita || pulita === 'CULTURA') return;
    try {
      await fetch(`${SUPABASE_CONFIG.url}/rest/v1/categorie`, {
        method: 'POST',
        headers: {
          ...headersSupabase(true),
          'Prefer': 'resolution=ignore-duplicates,return=representation'
        },
        body: JSON.stringify({ nome: pulita })
      });
    } catch (e) {}
  },

  async incrementaLettura(articolo) {
    if (!articolo || normalizzaStatoArticolo(articolo.stato_articolo) !== 'pubblicato') return;
    const norm = normalizzaArticolo(articolo);
    norm.letture += 1;
    norm.letture_oggi += 1;
    norm.letture_7g += 1;
    norm.letture_30g += 1;

    articolo.letture = norm.letture;
    articolo.letture_oggi = norm.letture_oggi;
    articolo.letture_7g = norm.letture_7g;
    articolo.letture_30g = norm.letture_30g;

    if (typeof articolo.id === 'number' || /^\d+$/.test(String(articolo.id || ''))) {
      try {
        await fetch(`${SUPABASE_CONFIG.url}/rest/v1/articoli?id=eq.${encodeURIComponent(articolo.id)}`, {
          method: 'PATCH',
          headers: headersSupabase(false),
          body: JSON.stringify({
            letture: norm.letture,
            letture_oggi: norm.letture_oggi,
            letture_7g: norm.letture_7g,
            letture_30g: norm.letture_30g
          })
        });
      } catch (e) {}
    }
  }
};

function articoliPerPaginaDispositivo() {
  const w = window.innerWidth || document.documentElement.clientWidth || 1024;
  const isLandscape = window.matchMedia ? window.matchMedia('(orientation: landscape)').matches : w > window.innerHeight;
  if (w < 680) return 1;
  if (w >= 1024 && isLandscape) return 4;
  return 2;
}

function isStanzaCreataDaUtente(stanza, username) {
  if (!stanza || !username) return false;
  const creatore = String(stanza.creata_da || '').trim().toLowerCase().replace(/^@/, '');
  const target = String(username || '').trim().toLowerCase().replace(/^@/, '');

function articoliFiltrati() {
  const catAttiva = String(stato.categoriaAttiva || 'TUTTI').trim().toUpperCase();
  const stanzaAttiva = String(stato.stanzaAttiva || '').trim().toLowerCase();

  return stato.articoli.filter(a => {
    const isPubblicato = normalizzaStatoArticolo(a.stato_articolo) === 'pubblicato';
    if (!isPubblicato) return false;
    const catArt = String(a.categoria || '').trim().toUpperCase();
    const stanzaArt = String(a.stanza_nome || '').trim().toLowerCase();
    const okCategoria = catAttiva === 'TUTTI' || catArt === catAttiva;
    const okStanza = stanzaAttiva ? stanzaArt === stanzaAttiva : !stanzaArt;
    return okCategoria && okStanza;
  });
}

async function gestisciInvioNotaStanza(e) {
  e.preventDefault();
  if (!stato.stanzaAttiva) return;
  if (!stato.utenteCorrente) {
    apriModaleAccount('login');
    return;
  }

  const input = document.getElementById('inputNotaStanza');
  const btn = document.getElementById('btnInviaNotaStanza');
  const testo = input.value.trim();
  if (!testo) return;

  if (trovaParoleVietate(testo).length > 0) {
    alert('La nota contiene termini non consentiti.');
    return;
  }

  btn.disabled = true;
  btn.textContent = 'Invio...';

  const res = await BackendSupabase.inviaMessaggioStanza({
    stanza_nome: stato.stanzaAttiva,
    autore_username: stato.utenteCorrente.username,
    testo
  });

  btn.disabled = false;
  btn.textContent = 'Invia nota';

  if (!res.ok) {
    alert('Impossibile inviare la nota in questo momento.');
    return;
  }

  stato.messaggiStanza.unshift(res.messaggio);
  input.value = '';
  renderPianoLavoroStanza();
}

function renderCategorie() {
  stato.categorie = filtraCategorieSenzaCultura(stato.categorie);
  if (stato.categoriaAttiva === 'CULTURA') stato.categoriaAttiva = 'TUTTI';

  const bar = document.getElementById('barraCategorie');
  bar.innerHTML = '';
  const catAttiva = String(stato.categoriaAttiva || 'TUTTI').trim().toUpperCase();

  stato.categorie.forEach(rawCat => {
    const cat = String(rawCat || '').trim().toUpperCase();
    if (!cat || cat === 'CULTURA') return;
    const btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'cat-pill' + (catAttiva === cat ? ' attiva' : '');
    btn.textContent = cat;
    btn.onclick = () => {
      stato.categoriaAttiva = cat;
      stato.paginaCorrente = 0;
      renderCategorie();
      renderPaginaGiornale();
    };
    bar.appendChild(btn);
  });

  const selectCat = document.getElementById('inputCategoria');
  if (selectCat) {
    const scelte = stato.categorie
      .map(c => String(c || '').trim().toUpperCase())
      .filter(c => c && c !== 'TUTTI' && c !== 'CULTURA');
    selectCat.innerHTML = scelte.map(c => `<option value="${c}">${c}</option>`).join('');
  }

  const selectStanza = document.getElementById('inputStanzaArticolo');
  if (selectStanza) {
    const opzioni = ['<option value="">Giornale pubblico (nessuna stanza)</option>'];
    const nomiInseriti = new Set();
    stato.stanze.forEach(s => {
      if (!s || !s.nome) return;
      const sbloccata = stato.stanzeSbloccate.includes(s.nome) || stato.stanzaAttiva === s.nome || !s.password_stanza;
      if (sbloccata && !nomiInseriti.has(s.nome)) {
        nomiInseriti.add(s.nome);
        opzioni.push(`<option value="${escapeHtml(s.nome)}">Stanza: ${escapeHtml(s.nome)}</option>`);
      }
    });
    if (stato.stanzaAttiva && !nomiInseriti.has(stato.stanzaAttiva)) {
      opzioni.push(`<option value="${escapeHtml(stato.stanzaAttiva)}">Stanza: ${escapeHtml(stato.stanzaAttiva)}</option>`);
    }
    selectStanza.innerHTML = opzioni.join('');
    if (stato.stanzaAttiva) {
      selectStanza.value = stato.stanzaAttiva;
    }
  }
}

function escapeHtml(testo) {
  return String(testo || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

async function apriApprofondimentoArticolo(idArticolo) {
  const art = stato.articoli.find(item => String(item.id) === String(idArticolo));
  if (!art || normalizzaStatoArticolo(art.stato_articolo) !== 'pubblicato') return;

  if (art.stanza_nome) {
    const stanza = stato.stanze.find(s => s.nome === art.stanza_nome);
    const haPassword = Boolean(stanza && String(stanza.password_stanza || '').trim());
    if (stanza && haPassword && !stato.stanzeSbloccate.includes(stanza.nome)) {
      richiediAccessoStanzaPerNome(stanza.nome);
      return;
    }
  }

  stato.articoloApprofondito = art;
  stato.stepApprofondimento = 2;
  aggiornaVistaApprofondimento();
  window.scrollTo({ top: 0, behavior: 'instant' });

  const chiaveArt = String(art.id || art.titolo || '');
  if (chiaveArt && !stato.approfonditiInSessione.has(chiaveArt)) {
    stato.approfonditiInSessione.add(chiaveArt);
    await BackendSupabase.incrementaLettura(art);
    salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  }
}

async function avanzaApprofondimento() {

  if (stato.stepApprofondimento === 2) {
    stato.stepApprofondimento = 3;
    aggiornaVistaApprofondimento();
    window.scrollTo({ top: 0, behavior: 'instant' });
  }
}

  if (stato.paginaCorrente >= totalePagine) stato.paginaCorrente = totalePagine - 1;
  if (stato.paginaCorrente < 0) stato.paginaCorrente = 0;

  const griglia = document.getElementById('grigliaArticoliPagina');
  const navPagine = document.getElementById('navPagineArticolo');
  const etichettaFoglio = document.getElementById('etichettaFoglioGiornale');

  etichettaFoglio.textContent = stato.stanzaAttiva
    ? `FOGLIO STANZA • ${stato.stanzaAttiva.toUpperCase()}`
    : (stato.categoriaAttiva === 'TUTTI'
        ? 'FOGLIO DEL GIORNALE'
        : `FOGLIO • ${stato.categoriaAttiva}`);

  if (!lista.length) {
    navPagine.style.display = 'none';
    document.getElementById('contatorePagine').textContent = 'Pagina 0 / 0';
    griglia.className = 'griglia-articoli-pagina';
    griglia.innerHTML = `
      <div class="pagina-vuota-box">
        <h2 class="titolo-articolo" style="margin-bottom:8px;">${

            ? `Nessun articolo pubblicato nella stanza "${escapeHtml(stato.stanzaAttiva)}"`

                ? 'Nessun articolo pubblicato'
                : `Nessun articolo in ${escapeHtml(stato.categoriaAttiva)}`)
        }</h2>
        <p style="margin:0; color:var(--grigio);">
          ${stato.stanzaAttiva
            ? 'Usa il piano di lavoro qui sopra o "+ Nuovo articolo nella stanza" per aggiungere il primo articolo del progetto.'

      </div>

    aggiornaVistaApprofondimento();
    return;
  }

  const inizio = stato.paginaCorrente * perPagina;
  const articoliPagina = lista.slice(inizio, inizio + perPagina);

  griglia.className = 'griglia-articoli-pagina' + (perPagina === 4 ? ' pag-piena-quattro' : '');
  navPagine.style.display = totalePagine > 1 ? 'flex' : 'none';
  document.getElementById('contatorePagine').textContent = `Pagina ${stato.paginaCorrente + 1} / ${totalePagine}`;

    const posizioneGlobale = inizio + idx;
    const latoFoto = posizioneGlobale % 2 === 0 ? 'sinistra' : 'destra';
    const stileRitaglio = posizioneGlobale % 2 === 0 ? 'stile-rivista-a' : 'stile-rivista-b';
    const catArt = String(art.categoria || 'CRONACA').trim().toUpperCase();

    return `
      <section class="ritaglio-articolo ${stileRitaglio}">
        <div>
          <div class="ritaglio-meta">
            <span class="badge-cat">${escapeHtml(catArt)}</span>
            <span class="firma-giornale">${formattaFirmaAcuraDi(art)}</span>
          </div>
          <h2 class="ritaglio-titolo">${escapeHtml(art.titolo || '')}</h2>
          <div class="flusso-testo-foto">
            <img
              class="foto-avvolta ${latoFoto}"
              src="${escapeHtml(art.immagine || IMMAGINE_DEFAULT)}"
              alt="${escapeHtml(art.titolo || 'Immagine articolo')}"
            >
            <p class="ritaglio-estratto">${escapeHtml(art.livello1 || '')}</p>
          </div>
        </div>
        <div class="ritaglio-piede">
          <button type="button" class="btn btn-approfondisci" onclick="apriApprofondimentoArticolo('${escapeHtml(art.id)}')">
            Approfondisci →
          </button>
          <span class="contatore-pagine">Art. ${posizioneGlobale + 1} di ${lista.length}</span>
        </div>
      </section>
    `;
  }).join('');

  aggiornaVistaApprofondimento();
}

function cambiaPagina(delta) {

    renderPaginaGiornale();
    card.classList.remove('anim-uscita-sinistra', 'anim-uscita-destra');
  }, 150);
}

function getListaUtentiCompleta() {
  const mappa = new Map();
  stato.utenti.forEach(u => {
    if (u && u.username) mappa.set(u.username.toLowerCase(), u);
  });
  if (stato.utenteCorrente && stato.utenteCorrente.username) {
    const cur = stato.utenteCorrente.username.toLowerCase();
    mappa.set(cur, {
      username: stato.utenteCorrente.username,
      bio: stato.utenteCorrente.bio || 'Autore su ICEBERG.'
    });
  }
  stato.articoli.forEach(a => {
    const uname = (a.autore_username || String(a.autore || '').toLowerCase().replace(/[^a-z0-9_]+/g, '_')).trim();
    if (uname && !mappa.has(uname)) {
      mappa.set(uname, {
        username: uname,
        bio: `Autore attivo su ICEBERG (${a.autore || uname}).`
      });
    }
  });
  return Array.from(mappa.values());
}

function renderRisultatiRicerca() {
  const box = document.getElementById('risultatiRicerca');
  const q = stato.queryRicerca.trim().toLowerCase().replace(/^@/, '');
  const btnPulisci = document.getElementById('btnPulisciRicerca');
  btnPulisci.style.display = stato.queryRicerca.trim() ? 'inline-block' : 'none';

  if (!q) {
    box.classList.remove('visibile');
    box.innerHTML = '';
    return;
  }

  const utentiTrovati = getListaUtentiCompleta().filter(u =>
    String(u.username || '').toLowerCase().includes(q) ||
    String(u.bio || '').toLowerCase().includes(q)
  );

  const stanzeTrovate = stato.stanze.filter(s =>
    String(s.nome || '').toLowerCase().includes(q) ||
    String(s.descrizione || '').toLowerCase().includes(q) ||
    String(s.creata_da || '').toLowerCase().includes(q)
  );

  const articoliTrovati = stato.articoli.filter(a => {
    if (normalizzaStatoArticolo(a.stato_articolo) !== 'pubblicato') return false;
    return String(a.titolo || '').toLowerCase().includes(q) ||
      String(a.categoria || '').toLowerCase().includes(q) ||
      String(a.autore || '').toLowerCase().includes(q) ||
      String(a.autore_username || '').toLowerCase().includes(q) ||
      String(a.livello1 || '').toLowerCase().includes(q) ||
      String(a.livello2 || '').toLowerCase().includes(q) ||
      String(a.livello3 || '').toLowerCase().includes(q);
  });

  let html = '';

  html += `<div class="risultati-titolo-sezione">Utenti (${utentiTrovati.length})</div>`;
  if (!utentiTrovati.length) {
    html += `<p style="font-size:0.84rem; color:var(--grigio); margin:0 0 10px;">Nessun utente trovato per "${escapeHtml(stato.queryRicerca)}".</p>`;
  } else {
    utentiTrovati.forEach(u => {
      const iniziale = String(u.username || 'U').charAt(0).toUpperCase();
      html += `
        <div class="card-risultato">
          <div class="card-risultato-sx">
            <div class="avatar-iniziale">${escapeHtml(iniziale)}</div>
            <div>
              <strong>@${escapeHtml(u.username)}</strong><br>
              <small>${escapeHtml(u.bio || 'Profilo autore su ICEBERG')}</small>
            </div>
          </div>
          <button type="button" class="btn btn-piccolo" onclick="apriProfiloUtente('${escapeHtml(u.username)}')">Vedi profilo</button>
        </div>
      `;
    });
  }

  html += `<div class="risultati-titolo-sezione">Stanze (${stanzeTrovate.length})</div>`;
  if (!stanzeTrovate.length) {
    html += `<p style="font-size:0.84rem; color:var(--grigio); margin:0 0 10px;">Nessuna stanza trovata per "${escapeHtml(stato.queryRicerca)}".</p>`;
  } else {
    stanzeTrovate.forEach(s => {
      const haPassword = Boolean(String(s.password_stanza || '').trim());
      const sbloccata = !haPassword || stato.stanzeSbloccate.includes(s.nome);
      const badgeStanza = sbloccata
        ? '<span class="badge-cat">Piano di lavoro</span>'
        : '<span class="badge-cat">🔒 Password</span>';
      html += `
        <div class="card-risultato">
          <div class="card-risultato-sx">
            <img class="thumb-risultato" src="${escapeHtml(s.immagine || IMMAGINE_DEFAULT)}" alt="${escapeHtml(s.nome)}">
            <div>
              <strong>${escapeHtml(s.nome)}</strong> ${badgeStanza}<br>
              <small>${escapeHtml(s.descrizione || 'Spazio di progetto condiviso')} • ${escapeHtml(s.creata_da || '')}</small>
            </div>
          </div>
          <button type="button" class="btn btn-piccolo btn-primario" onclick="richiediAccessoStanzaPerNome('${escapeHtml(s.nome)}')">Entra</button>
        </div>
      `;
    });
  }

  html += `<div class="risultati-titolo-sezione">Articoli (${articoliTrovati.length})</div>`;
  if (!articoliTrovati.length) {
    html += `<p style="font-size:0.84rem; color:var(--grigio); margin:0;">Nessun articolo trovato per "${escapeHtml(stato.queryRicerca)}".</p>`;
  } else {
    articoliTrovati.forEach(a => {
      html += `
        <div class="card-risultato">
          <div class="card-risultato-sx">
            <img class="thumb-risultato" src="${escapeHtml(a.immagine || IMMAGINE_DEFAULT)}" alt="${escapeHtml(a.titolo)}">
            <div>
              <strong>${escapeHtml(a.titolo)}</strong><br>
              <small>${escapeHtml(a.categoria)} • A cura di @${escapeHtml(a.autore_username || a.autore)}${a.stanza_nome ? ' • Stanza: ' + escapeHtml(a.stanza_nome) : ''}</small>
            </div>
          </div>
          <button type="button" class="btn btn-piccolo" onclick="apriArticoloDaRicerca('${escapeHtml(a.id)}')">Apri</button>
        </div>
      `;
    });
  }

  box.innerHTML = html;
  box.classList.add('visibile');
}

  const tuttiUtenti = getListaUtentiCompleta();

    username: clean || 'autore',
    bio: 'Profilo su ICEBERG.'
  };

  stato.profiloVisualizzato = profilo.username;

  const isMioProfilo = Boolean(
    stato.utenteCorrente &&
    String(stato.utenteCorrente.username || '').toLowerCase() === profilo.username.toLowerCase()
  );

  const iniziale = String(profilo.username || 'U').charAt(0).toUpperCase();
  document.getElementById('titoloProfiloUtente').textContent = isMioProfilo
    ? `Il mio profilo (@${profilo.username})`
    : `Profilo @${profilo.username}`;

  const azioniMioProfilo = isMioProfilo ? `
    <div class="riga-flex" style="margin-top:14px;">
      <button type="button" class="btn btn-piccolo btn-primario" onclick="chiudiModali(); apriModaleAccount('login');">Modifica nome profilo, bio o password</button>
      <button type="button" class="btn btn-piccolo btn-secondario" onclick="chiudiModali(); apriModaleAutore('pubblicati');">Vai all'Area Autore</button>
    </div>
  ` : '';

  document.getElementById('contenutoProfiloUtente').innerHTML = `
    <div class="stat-card">
      <div class="card-risultato-sx">
        <div class="avatar-iniziale">${escapeHtml(iniziale)}</div>
        <div>
          <strong>@${escapeHtml(profilo.username)}</strong><br>
          <small>${escapeHtml(profilo.bio || 'Profilo su ICEBERG')}</small>

        ? 'Questa sezione raccoglie i dati del tuo profilo e le impostazioni di accesso (nome profilo, bio e password). Per articoli, bozze, stanze e statistiche apri l’Area Autore.'

  document.getElementById('modaleProfilo').classList.add('aperto');

function trovaStanzaByKey(key) {
  const target = String(key || '');
  return stato.stanze.find(s =>
    (s.id !== undefined && s.id !== null && String(s.id) === target) ||
    String(s.nome || '') === target
  );
}

async function entraInStanzaByKey(key) {
  const stanza = trovaStanzaByKey(key);
  if (stanza) {
    await entraInStanza(stanza);
  }
}

function apriModificaStanza(key) {
  if (!stato.utenteCorrente) return;
  const stanza = trovaStanzaByKey(key);
  if (!stanza || !isStanzaCreataDaUtente(stanza, stato.utenteCorrente.username)) {
    alert('Puoi modificare solo le stanze create personalmente dal tuo account.');
    return;
  }

  chiudiModali();
  stato.immagineModStanzaDataUrl = '';
  document.getElementById('editStanzaId').value = stanza.id !== undefined && stanza.id !== null ? String(stanza.id) : '';
  document.getElementById('editStanzaNomeOriginale').value = stanza.nome || '';
  document.getElementById('modNomeStanza').value = stanza.nome || '';
  document.getElementById('modPasswordStanza').value = stanza.password_stanza || '';
  document.getElementById('modDescrizioneStanza').value = stanza.descrizione || '';
  document.getElementById('modImmagineStanzaUrl').value = (stanza.immagine && !stanza.immagine.startsWith('data:')) ? stanza.immagine : '';
  document.getElementById('dropzoneModStanzaImg').textContent = 'Trascina qui una nuova immagine o tocca per sceglierla';
  document.getElementById('boxControlloModStanza').className = 'avviso-ai';
  document.getElementById('modaleModificaStanza').classList.add('aperto');
}

function valutaControlloModStanza() {
  const testo = [
    document.getElementById('modNomeStanza').value,
    document.getElementById('modDescrizioneStanza').value
  ].join(' ');
  const box = document.getElementById('boxControlloModStanza');
  if (!testo.trim()) {
    box.className = 'avviso-ai';
    return [];
  }
  const vietate = trovaParoleVietate(testo);
  if (vietate.length > 0) {
    box.className = 'avviso-ai allerta';
    box.textContent = 'Controllo linguaggio: rilevati termini non consentiti nel nome o nella descrizione della stanza.';
  } else {
    box.className = 'avviso-ai ok';
    box.textContent = 'Controllo linguaggio: nessun termine vietato rilevato.';
  }
  return vietate;
}

['modNomeStanza', 'modDescrizioneStanza'].forEach(id => {
  document.getElementById(id).addEventListener('input', valutaControlloModStanza);
});

document.getElementById('formModificaStanza').addEventListener('submit', async e => {
  e.preventDefault();
  if (!stato.utenteCorrente) return;
  if (valutaControlloModStanza().length > 0) return;

  const nomeOriginale = document.getElementById('editStanzaNomeOriginale').value;
  const idx = stato.stanze.findIndex(s =>
    String(s.nome || '') === String(nomeOriginale || '') &&
    isStanzaCreataDaUtente(s, stato.utenteCorrente.username)
  );
  if (idx < 0) return;

  const stanzaOriginale = stato.stanze[idx];
  const nuovoNome = document.getElementById('modNomeStanza').value.trim();
  const nuovaPassword = document.getElementById('modPasswordStanza').value;
  const nuovaDescrizione = document.getElementById('modDescrizioneStanza').value.trim();
  const urlImg = document.getElementById('modImmagineStanzaUrl').value.trim();
  const immagineFinale = urlImg || stato.immagineModStanzaDataUrl || stanzaOriginale.immagine || IMMAGINE_DEFAULT;

  if (!nuovoNome) return;

  const btn = document.getElementById('btnSalvaModStanza');
  btn.disabled = true;
  btn.textContent = 'Salvataggio...';

  const aggiornata = await BackendSupabase.aggiornaStanza(stanzaOriginale, {
    nome: nuovoNome,
    password_stanza: nuovaPassword,
    descrizione: nuovaDescrizione,
    immagine: immagineFinale
  });

  if (nomeOriginale && nomeOriginale !== nuovoNome) {
    await BackendSupabase.rinominaRiferimentiStanza(nomeOriginale, nuovoNome);
    stato.articoli.forEach(a => {
      if (String(a.stanza_nome || '') === String(nomeOriginale)) {
        a.stanza_nome = nuovoNome;
      }
    });
    if (stato.stanzeSbloccate.includes(nomeOriginale)) {
      stato.stanzeSbloccate = stato.stanzeSbloccate.map(n => n === nomeOriginale ? nuovoNome : n);
      salvaStanzeSbloccate(stato.stanzeSbloccate);
    }
    if (String(stato.stanzaAttiva || '') === String(nomeOriginale)) {
      stato.stanzaAttiva = nuovoNome;
      stato.messaggiStanza = await BackendSupabase.caricaMessaggiStanza(nuovoNome);
    }
  }

  stato.stanze[idx] = {
    ...stanzaOriginale,
    ...aggiornata,
    nome: nuovoNome,
    password_stanza: nuovaPassword,
    descrizione: nuovaDescrizione,
    immagine: immagineFinale
  };

  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  stato.immagineModStanzaDataUrl = '';
  btn.disabled = false;
  btn.textContent = 'Salva impostazioni stanza';

  renderCategorie();
  renderPaginaGiornale();
  if (stato.queryRicerca) renderRisultatiRicerca();
  apriModaleAutore('stanze');
});

function eliminaStanzaUtente(key) {
  if (!stato.utenteCorrente) return;
  const stanza = trovaStanzaByKey(key);
  if (!stanza || !isStanzaCreataDaUtente(stanza, stato.utenteCorrente.username)) {
    alert('Puoi eliminare solo le stanze create personalmente dal tuo account.');
    return;
  }

  apriConfermaEliminazione({
    titolo: 'Elimina stanza',
    occhio: 'STANZA DI PROGETTO • CONFERMA RICHIESTA',
    domanda: `Vuoi davvero eliminare la stanza "${stanza.nome}"?`,
    nota: 'La stanza verrà rimossa dall’archivio e gli articoli associati torneranno sul giornale generale.',
    etichettaConferma: 'Sì, elimina stanza',
    onConferma: async () => {
      await BackendSupabase.eliminaStanza(stanza);

      const nomeEliminato = stanza.nome;
      stato.stanze = stato.stanze.filter(s => s !== stanza);
      stato.articoli.forEach(a => {
        if (String(a.stanza_nome || '').toLowerCase() === String(nomeEliminato || '').toLowerCase()) {
          a.stanza_nome = '';
        }
      });
      stato.stanzeSbloccate = stato.stanzeSbloccate.filter(n => n !== nomeEliminato);
      salvaStanzeSbloccate(stato.stanzeSbloccate);

      if (String(stato.stanzaAttiva || '').toLowerCase() === String(nomeEliminato || '').toLowerCase()) {
        stato.stanzaAttiva = '';
        stato.messaggiStanza = [];
      }

      salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
      renderCategorie();
      renderPaginaGiornale();
      if (stato.queryRicerca) renderRisultatiRicerca();
      if (document.getElementById('modaleAutore').classList.contains('aperto') && stato.utenteCorrente) {
        apriModaleAutore('stanze');
      }
    }
  });
}

async function cambiaStatoArticolo(idArticolo, nuovoStato) {
  const idx = stato.articoli.findIndex(a => String(a.id) === String(idArticolo));
  if (idx < 0) return;

  const statoPulito = normalizzaStatoArticolo(nuovoStato);
  stato.articoli[idx].stato_articolo = statoPulito;

  await BackendSupabase.aggiornaArticolo(stato.articoli[idx].id, {
    stato_articolo: statoPulito
  });

  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  renderPaginaGiornale();
  if (stato.queryRicerca) renderRisultatiRicerca();

  if (document.getElementById('modaleAutore').classList.contains('aperto') && stato.utenteCorrente) {
    const schedaRitorno = statoPulito === 'archiviato' ? 'archiviati' : 'pubblicati';
    apriModaleAutore(schedaRitorno);
  }
}

function eliminaArticoloUtente(idArticolo) {
  const idx = stato.articoli.findIndex(a => String(a.id) === String(idArticolo));
  if (idx < 0) return;
  const art = stato.articoli[idx];

  apriConfermaEliminazione({
    titolo: 'Elimina articolo',
    occhio: 'REDAZIONE ICEBERG • CONFERMA ELIMINAZIONE',
    domanda: `Vuoi davvero eliminare l’articolo "${art.titolo}"?`,
    nota: 'L’articolo e le relative statistiche verranno eliminati in modo permanente.',
    etichettaConferma: 'Sì, elimina articolo',
    onConferma: async () => {
      const idxCorrente = stato.articoli.findIndex(a => String(a.id) === String(idArticolo));
      if (idxCorrente < 0) return;
      const target = stato.articoli[idxCorrente];

      await BackendSupabase.eliminaArticolo(target.id);
      stato.articoli.splice(idxCorrente, 1);

      const chiaveArt = String(target.id || target.titolo || '');
      if (chiaveArt) stato.approfonditiInSessione.delete(chiaveArt);

      salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
      renderPaginaGiornale();
      if (stato.queryRicerca) renderRisultatiRicerca();
      if (document.getElementById('modaleAutore').classList.contains('aperto') && stato.utenteCorrente) {
        apriModaleAutore(stato.schedaAreaAutore || 'pubblicati');
      }
    }
  });
}

function pulisciRicerca() {

  document.getElementById('inputRicerca').value = '';

}

function richiediAccessoStanzaPerNome(nomeStanza) {
  const stanza = stato.stanze.find(s => String(s.nome || '').toLowerCase() === String(nomeStanza || '').toLowerCase());
  if (!stanza) return;

  const haPassword = Boolean(String(stanza.password_stanza || '').trim());
  if (!haPassword || stato.stanzeSbloccate.includes(stanza.nome)) {
    entraInStanza(stanza);
    return;
  }

  stato.stanzaInAccesso = stanza;
  document.getElementById('titoloAccessoStanza').textContent = `Stanza: ${stanza.nome}`;
  document.getElementById('descAccessoStanza').textContent = stanza.descrizione || 'Inserisci la password per sbloccare ed entrare nel piano di lavoro.';
  document.getElementById('inputPasswordAccesso').value = '';
  document.getElementById('errorePasswordStanza').style.display = 'none';
  document.getElementById('modaleAccessoStanza').classList.add('aperto');
}

function richiediAccessoStanza(indiceStanza) {
  const stanza = stato.stanze[indiceStanza];
  if (!stanza) return;
  richiediAccessoStanzaPerNome(stanza.nome);
}

document.getElementById('formAccessoStanza').addEventListener('submit', e => {
  e.preventDefault();
  const stanza = stato.stanzaInAccesso;
  if (!stanza) return;
  const pwd = document.getElementById('inputPasswordAccesso').value;
  if (pwd === stanza.password_stanza) {
    if (!stato.stanzeSbloccate.includes(stanza.nome)) {
      stato.stanzeSbloccate.push(stanza.nome);
      salvaStanzeSbloccate(stato.stanzeSbloccate);
    }
    stato.stanzaInAccesso = null;
    chiudiModali();
    entraInStanza(stanza);
  } else {
    document.getElementById('errorePasswordStanza').style.display = 'block';
  }
});

async function entraInStanza(stanza) {
  if (!stanza || !stanza.nome) return;
  stato.stanzaAttiva = stanza.nome;
  stato.categoriaAttiva = 'TUTTI';
  stato.paginaCorrente = 0;
  stato.stepApprofondimento = 1;
  stato.articoloApprofondito = null;
  stato.messaggiStanza = await BackendSupabase.caricaMessaggiStanza(stanza.nome);
  pulisciRicerca();
  renderCategorie();
  renderPaginaGiornale();
  document.getElementById('pianoLavoroStanza').scrollIntoView({ behavior: 'smooth', block: 'start' });
}

async function apriArticoloDaRicerca(idArticolo) {
  const art = stato.articoli.find(item => String(item.id) === String(idArticolo));
  if (!art || normalizzaStatoArticolo(art.stato_articolo) !== 'pubblicato') return;
  pulisciRicerca();
  await apriApprofondimentoArticolo(art.id);
}

document.getElementById('inputRicerca').addEventListener('input', e => {
  stato.queryRicerca = e.target.value;
  renderRisultatiRicerca();
});

let touchStartX = 0;
let touchStartY = 0;
const lettoreEl = document.querySelector('.lettore-wrap');
lettoreEl.addEventListener('touchstart', e => {
  if (stato.stepApprofondimento < 2 && e.touches.length === 1) {
    touchStartX = e.touches[0].clientX;
    touchStartY = e.touches[0].clientY;
  }
}, { passive: true });

lettoreEl.addEventListener('touchend', e => {
  if (stato.stepApprofondimento >= 2 || !e.changedTouches.length) return;
  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;
  if (Math.abs(dx) > 55 && Math.abs(dx) > Math.abs(dy) * 1.4) {
    if (dx < 0) cambiaPagina(1);
    else cambiaPagina(-1);
  }
}, { passive: true });

function apriModaleAccount(tab = 'login', azioneDopo = null) {

  const authBox = document.getElementById('pannelloAuthForms');
  const msgBox = document.getElementById('messaggioGestioneAccount');

  if (stato.utenteCorrente) {
    loggatoBox.style.display = 'block';

function completaAccessoUtente(utente) {
  stato.utenteCorrente = utente;
  if (!stato.utenti.some(u => u.username.toLowerCase() === utente.username.toLowerCase())) {
    stato.utenti.unshift({ username: utente.username, bio: utente.bio || '' });
  }
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  aggiornaBottoneAccount();
  const azione = stato.azioneDopoLogin;
  stato.azioneDopoLogin = null;
  chiudiModali();
  if (typeof azione === 'function') {
    azione();
  } else if (stato.stanzaAttiva) {
    renderPianoLavoroStanza();
  }
}

document.getElementById('formLogin').addEventListener('submit', async e => {
  e.preventDefault();
  const email = document.getElementById('loginEmail').value.trim().toLowerCase();
  const password = document.getElementById('loginPassword').value;
  const errBox = document.getElementById('erroreAuth');
  const btn = document.getElementById('btnSubmitLogin');

  btn.disabled = true;
  btn.textContent = 'Accesso...';
  errBox.style.display = 'none';

  const res = await BackendSupabase.loginConAuth({ email, password });

  btn.disabled = false;
  btn.textContent = 'Accedi';

  if (!res.ok) {

    errBox.style.display = 'block';

document.getElementById('formRegister').addEventListener('submit', async e => {
  e.preventDefault();
  const rawUser = document.getElementById('regUsername').value;
  const bio = document.getElementById('regBio').value.trim();
  const vietate = trovaParoleVietate(`${rawUser} ${bio}`);
  const errBox = document.getElementById('erroreRegister');

  if (vietate.length > 0) {
    errBox.style.color = 'var(--allerta)';
    errBox.textContent = 'Nome utente o bio contengono parole non consentite.';
    errBox.style.display = 'block';
    return;
  }

  const username = pulisciUsername(rawUser, document.getElementById('regEmail').value);
  const email = document.getElementById('regEmail').value.trim().toLowerCase();
  const password = document.getElementById('regPassword').value;
  const btn = document.getElementById('btnSubmitRegister');

  if (password.length < 6) {
    errBox.style.color = 'var(--allerta)';
    errBox.textContent = 'Inserisci una password di almeno 6 caratteri.';
    errBox.style.display = 'block';
    return;
  }

  btn.disabled = true;
  btn.textContent = 'Registrazione...';
  errBox.style.display = 'none';

  const res = await BackendSupabase.registraConAuth({ username, email, password, bio });

  btn.disabled = false;
  btn.textContent = 'Registrati';

  if (!res.ok) {
    errBox.style.color = 'var(--allerta)';
    errBox.textContent = res.errore;
    errBox.style.display = 'block';
    return;
  }

  if (!stato.utenti.some(u => u.username.toLowerCase() === res.utente.username.toLowerCase())) {
    stato.utenti.unshift({ username: res.utente.username, bio: res.utente.bio || '' });
    salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  }

  e.target.reset();

  if (res.richiedeConfermaEmail) {
    errBox.style.color = 'var(--ok)';
    errBox.textContent = 'Account creato. Controlla la tua email per confermarlo, poi accedi da "Accedi".';
    errBox.style.display = 'block';
    return;
  }

  completaAccessoUtente(res.utente);
});

async function salvaImpostazioniAccount() {
  if (!stato.utenteCorrente) return;
  const msgBox = document.getElementById('messaggioGestioneAccount');
  const vecchioUsername = stato.utenteCorrente.username;
  const rawUser = document.getElementById('inputMioUsername').value;
  const nuovaBio = document.getElementById('inputMiaBio').value.trim();
  const nuovaPwd = document.getElementById('inputNuovaPassword').value;

  const vietate = trovaParoleVietate(`${rawUser} ${nuovaBio}`);
  if (vietate.length > 0) {
    msgBox.style.display = 'block';
    msgBox.style.color = 'var(--allerta)';
    msgBox.textContent = 'Nome utente o bio contengono parole non consentite.';
    return;
  }

  const nuovoUsername = pulisciUsername(rawUser, stato.utenteCorrente.email);
  if (nuovaPwd && nuovaPwd.length < 6) {
    msgBox.style.display = 'block';
    msgBox.style.color = 'var(--allerta)';
    msgBox.textContent = 'La nuova password deve avere almeno 6 caratteri.';
    return;
  }

  const btn = document.getElementById('btnSalvaAccount');
  btn.disabled = true;
  btn.textContent = 'Salvataggio...';

  const resAuth = await BackendSupabase.aggiornaUtenteAuth({
    username: nuovoUsername,
    bio: nuovaBio,
    password: nuovaPwd
  });

  btn.disabled = false;
  btn.textContent = 'Salva impostazioni profilo';

  if (!resAuth.ok) {
    msgBox.style.display = 'block';
    msgBox.style.color = 'var(--allerta)';
    msgBox.textContent = resAuth.errore;
    return;
  }

  stato.utenteCorrente.username = nuovoUsername;
  stato.utenteCorrente.bio = nuovaBio;

  const idxUtente = stato.utenti.findIndex(u => u.username.toLowerCase() === vecchioUsername.toLowerCase());
  if (idxUtente >= 0) {
    stato.utenti[idxUtente] = { username: nuovoUsername, bio: nuovaBio };
  } else {
    stato.utenti.unshift({ username: nuovoUsername, bio: nuovaBio });
  }

  if (vecchioUsername !== nuovoUsername) {
    stato.articoli.forEach(a => {
      if (String(a.autore_username || '').toLowerCase() === vecchioUsername.toLowerCase()) {
        a.autore_username = nuovoUsername;
        a.autore = nuovoUsername;
      }
    });
    stato.stanze.forEach(s => {
      if (isStanzaCreataDaUtente(s, vecchioUsername)) {
        s.creata_da = `@${nuovoUsername}`;
      }
    });
  }

  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  await BackendSupabase.aggiornaProfiloTabella(vecchioUsername, nuovoUsername, nuovaBio);
  aggiornaBottoneAccount();
  renderPaginaGiornale();

  msgBox.style.display = 'block';
  msgBox.style.color = 'var(--ok)';
  msgBox.textContent = nuovaPwd
    ? 'Profilo e password aggiornati.'
    : 'Profilo aggiornato.';
  document.getElementById('inputNuovaPassword').value = '';
  document.getElementById('mioUsername').textContent = `@${nuovoUsername}`;
}

    .map(normalizzaArticolo)
    .filter(a => {
      const autUser = String(a.autore_username || '').trim().toLowerCase();

  const tot30g = pubblicati.reduce((acc, a) => acc + a.letture_30g, 0);
  const totStorico = pubblicati.reduce((acc, a) => acc + a.letture, 0);

async function confermaNuovaCategoria() {
  const input = document.getElementById('inputNuovaCategoria');
  const nome = input.value;
  if (!nome || !nome.trim()) {
    input.focus();
    return;
  }
  const pulita = nome.trim().toUpperCase();
  if (pulita === 'CULTURA' || pulita === 'TUTTI') {
    alert('Nome categoria non disponibile.');
    return;
  }
  if (trovaParoleVietate(pulita).length > 0) {
    alert('Nome categoria non consentito.');
    return;
  }
  if (!stato.categorie.includes(pulita)) {
    stato.categorie.push(pulita);
  }
  await BackendSupabase.creaCategoria(pulita);
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
  renderCategorie();
  document.getElementById('inputCategoria').value = pulita;
  chiudiBoxNuovaCategoria();
}

function collegaDropzone(dropEl, inputFileEl, inputUrlEl, onCaricata) {
  function leggiFile(file) {
    if (!file || !file.type.startsWith('image/')) return;
    const reader = new FileReader();
    reader.onload = ev => {
      onCaricata(ev.target.result, file.name);
      if (inputUrlEl) inputUrlEl.value = '';
    };
    reader.readAsDataURL(file);
  }

  dropEl.addEventListener('dragover', e => {
    e.preventDefault();
    dropEl.classList.add('dragover');
  });
  dropEl.addEventListener('dragleave', () => dropEl.classList.remove('dragover'));
  dropEl.addEventListener('drop', e => {
    e.preventDefault();
    dropEl.classList.remove('dragover');
    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
      leggiFile(e.dataTransfer.files[0]);
    }
  });
  inputFileEl.addEventListener('change', e => {
    if (e.target.files && e.target.files[0]) {
      leggiFile(e.target.files[0]);
    }
  });
}

    'non esitare a chiedere'
  ];
  const trovati = indiziAi.filter(frase => testoCompleto.includes(frase));

  if (trovati.length > 0) {
    box.className = 'avviso-ai allerta';
    box.textContent = `Controllo anti-AI: rilevate formule tipiche da assistente (${trovati.join(', ')}). Riscrivi in modo più naturale e personale.`;
  } else {
    box.className = 'avviso-ai ok';
    box.textContent = 'Controllo anti-AI: nessun pattern tipico rilevato.';
  }
}

function valutaControlloParoleArticolo() {
  const testoCompleto = [
    document.getElementById('inputTitolo').value,
    document.getElementById('inputLiv1').value,
    document.getElementById('inputLiv2').value,
    document.getElementById('inputLiv3').value
  ].join(' ');

  const box = document.getElementById('boxControlloParole');
  if (!testoCompleto.trim()) {
    box.className = 'avviso-ai';
    return [];
  }

  const vietate = trovaParoleVietate(testoCompleto);
  if (vietate.length > 0) {
    box.className = 'avviso-ai allerta';
    box.textContent = 'Controllo linguaggio: rilevati termini discriminatori o offensivi non consentiti. Rimuovili prima di salvare o pubblicare.';
  } else {
    box.className = 'avviso-ai ok';
    box.textContent = 'Controllo linguaggio: nessun termine vietato rilevato.';
  }
  return vietate;
}

function valutaControlloParoleStanza() {
  const testoStanza = [
    document.getElementById('inputNomeStanza').value,
    document.getElementById('inputDescrizioneStanza').value
  ].join(' ');
  const box = document.getElementById('boxControlloStanza');
  if (!testoStanza.trim()) {
    box.className = 'avviso-ai';
    return [];
  }
  const vietate = trovaParoleVietate(testoStanza);
  if (vietate.length > 0) {
    box.className = 'avviso-ai allerta';
    box.textContent = 'Controllo linguaggio: rilevati termini non consentiti nel nome o nella descrizione della stanza.';
  } else {
    box.className = 'avviso-ai ok';
    box.textContent = 'Controllo linguaggio: nessun termine vietato rilevato.';
  }
  return vietate;
}

['inputTitolo', 'inputLiv1', 'inputLiv2', 'inputLiv3'].forEach(id => {
  document.getElementById(id).addEventListener('input', () => {
    valutaControlloAi();
    valutaControlloParoleArticolo();
  });
});

['inputNomeStanza', 'inputDescrizioneStanza'].forEach(id => {
  document.getElementById(id).addEventListener('input', valutaControlloParoleStanza);
});

async function gestisciSalvataggioArticolo(statoDestinazione = 'pubblicato') {
  if (!stato.utenteCorrente) {
    chiudiModali();
    apriModaleAccount('login');
    return;
  }

  const titolo = document.getElementById('inputTitolo').value.trim();
  const liv1 = document.getElementById('inputLiv1').value.trim();
  const liv2 = document.getElementById('inputLiv2').value.trim();
  const liv3 = document.getElementById('inputLiv3').value.trim();

  if (!titolo || !liv1 || !liv2 || !liv3) {
    alert('Compila titolo e tutti e tre i livelli prima di procedere.');
    return;
  }

  const vietate = valutaControlloParoleArticolo();
  if (vietate.length > 0) {
    return;
  }

  const statoFinale = normalizzaStatoArticolo(statoDestinazione);
  const urlInserito = document.getElementById('inputImmagineUrl').value.trim();
  const immagineFinale = urlInserito || stato.immagineArticoloDataUrl || IMMAGINE_DEFAULT;
  const stanzaScelta = document.getElementById('inputStanzaArticolo').value;
  const categoriaScelta = document.getElementById('inputCategoria').value || 'CRONACA';
  const editId = document.getElementById('editArticoloId').value;

  const btnPubblica = document.getElementById('btnPubblica');
  const btnBozza = document.getElementById('btnSalvaBozza');
  btnPubblica.disabled = true;
  btnBozza.disabled = true;
  btnPubblica.textContent = 'Salvataggio...';

  try {
    if (editId) {
      const idxEsistente = stato.articoli.findIndex(a => String(a.id) === String(editId));
      if (idxEsistente >= 0) {
        const corrente = stato.articoli[idxEsistente];
        const aggiornato = {
          ...corrente,
          categoria: categoriaScelta,
          stanza_nome: stanzaScelta,
          immagine: immagineFinale,
          titolo,
          livello1: liv1,
          livello2: liv2,
          livello3: liv3,
          stato_articolo: statoFinale
        };
        await BackendSupabase.aggiornaArticolo(corrente.id, {
          categoria: aggiornato.categoria,
          stanza_nome: aggiornato.stanza_nome,
          immagine: aggiornato.immagine,
          titolo: aggiornato.titolo,
          livello1: aggiornato.livello1,
          livello2: aggiornato.livello2,
          livello3: aggiornato.livello3,
          stato_articolo: aggiornato.stato_articolo
        });
        stato.articoli[idxEsistente] = aggiornato;
      }
    } else {
      const creato = await BackendSupabase.creaArticolo({
        categoria: categoriaScelta,
        stanza_nome: stanzaScelta,
        autore: stato.utenteCorrente.username,
        autore_username: stato.utenteCorrente.username,
        immagine: immagineFinale,
        titolo,
        livello1: liv1,
        livello2: liv2,
        livello3: liv3,
        stato_articolo: statoFinale
      });
      stato.articoli.unshift(creato);
    }

    salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);
    document.getElementById('editArticoloId').value = '';
    stato.immagineArticoloDataUrl = '';
    document.getElementById('dropzoneImg').textContent = 'Trascina qui un\'immagine o tocca per sceglierla dal dispositivo';
    chiudiBoxNuovaCategoria();
    document.getElementById('formArticolo').reset();
    document.getElementById('boxControlloAi').className = 'avviso-ai';
    document.getElementById('boxControlloParole').className = 'avviso-ai';

    chiudiModali();

    if (editId) {
      renderPaginaGiornale();
      if (stato.queryRicerca) renderRisultatiRicerca();
      apriModaleAutore(statoFinale === 'bozza' ? 'bozze' : (statoFinale === 'archiviato' ? 'archiviati' : 'pubblicati'));
    } else if (statoFinale === 'bozza') {
      apriModaleAutore('bozze');
    } else {
      stato.stanzaAttiva = stanzaScelta || '';
      stato.categoriaAttiva = 'TUTTI';
      stato.paginaCorrente = 0;
      renderCategorie();
      renderPaginaGiornale();
      if (stato.queryRicerca) renderRisultatiRicerca();
    }
  } catch (err) {
    alert('Salvataggio non riuscito: ' + (err && err.message ? err.message : 'errore imprevisto.'));
  } finally {
    btnPubblica.disabled = false;
    btnBozza.disabled = false;
    btnPubblica.textContent = 'Pubblica sul giornale';
  }
}

document.getElementById('formArticolo').addEventListener('submit', async e => {
  e.preventDefault();
  await gestisciSalvataggioArticolo('pubblicato');
});

document.getElementById('formStanza').addEventListener('submit', async e => {
  e.preventDefault();
  if (!stato.utenteCorrente) {
    chiudiModali();
    apriModaleAccount('login');
    return;
  }

  if (valutaControlloParoleStanza().length > 0) {
    return;
  }

  const nomeStanza = document.getElementById('inputNomeStanza').value.trim();
  const passwordStanza = document.getElementById('inputPasswordStanza').value;
  if (!nomeStanza) return;

  const urlStanza = document.getElementById('inputImmagineStanzaUrl').value.trim();
  const immagineStanza = urlStanza || stato.immagineStanzaDataUrl || IMMAGINE_DEFAULT;

  const btnCrea = document.getElementById('btnCreaStanza');
  btnCrea.disabled = true;
  btnCrea.textContent = 'Creazione...';

  const creata = await BackendSupabase.creaStanza({
    nome: nomeStanza,
    password_stanza: passwordStanza,
    creata_da: `@${stato.utenteCorrente.username}`,
    descrizione: document.getElementById('inputDescrizioneStanza').value,
    immagine: immagineStanza
  });

  stato.stanze.unshift(creata);
  if (!stato.stanzeSbloccate.includes(creata.nome)) {
    stato.stanzeSbloccate.push(creata.nome);
    salvaStanzeSbloccate(stato.stanzeSbloccate);
  }
  salvaCacheLocale(stato.articoli, stato.categorie, stato.stanze, stato.utenti);

  stato.immagineStanzaDataUrl = '';
  document.getElementById('dropzoneStanzaImg').textContent = 'Trascina qui un\'immagine per la stanza o tocca per sceglierla';
  document.getElementById('boxControlloStanza').className = 'avviso-ai';
  e.target.reset();

  btnCrea.disabled = false;
  btnCrea.textContent = 'Crea stanza';
  chiudiModali();
  await entraInStanza(creata);
  if (stato.queryRicerca) renderRisultatiRicerca();
});
